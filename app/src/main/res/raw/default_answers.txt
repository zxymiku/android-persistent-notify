【题库1-第1题】
题目：用 for 循环计算 1 到 100 的整数和，并输出结果。
答案：
#include <stdio.h>
int main(void)
{
    int i, sum = 0;
    for (i = 1; i <= 100; i++)
        sum += i;
    printf("1到100的和为：%d\n", sum);
    return 0;
}

【题库1-第20题】
题目：用 while 循环实现 "猜数字游戏"：程序随机生成 1-100 的整数，用户每次输入猜测值，程序提示 "太大""太小"，直到猜对为止，统计猜测次数。
答案：
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
int main(void)
{
    int target, guess, count = 0;
    srand((unsigned)time(NULL));
    target = rand() % 100 + 1;
    printf("猜数字游戏开始（1-100）！\n");
    while (1) {
        printf("请输入你猜的数字：");
        scanf("%d", &guess);
        count++;
        if (guess > target)
            printf("太大\n");
        else if (guess < target)
            printf("太小\n");
        else {
            printf("恭喜你猜对了！答案是%d，共猜了%d次。\n", target, count);
            break;
        }
    }
    return 0;
}

答案：
#include <stdio.h>
int main(void)
{
    int num, max, min, first = 1;
    printf("请输入多个整数（输入-1停止）：");
    while (1) {
        scanf("%d", &num);
        if (num == -1)
            break;
        if (first) {
            max = num;
            min = num;
            first = 0;
        } else {
            if (num > max)
                max = num;
            if (num < min)
                min = num;
        }
    }
    if (first)
        printf("没有输入有效数据。\n");
    else {
        printf("最大值为：%d\n", max);
        printf("最小值为：%d\n", min);
    }
    return 0;
}

{
    int a, b;
    char op;
    printf("请输入表达式（如 3+5）：");
    scanf("%d %c %d", &a, &op, &b);
    switch (op) {
    case '+':
        printf("%d %c %d = %d\n", a, op, b, a + b);
        break;
    case '-':
        printf("%d %c %d = %d\n", a, op, b, a - b);
        break;
    case '*':
        printf("%d %c %d = %d\n", a, op, b, a * b);
        break;
    case '/':
        if (b != 0)
            printf("%d %c %d = %d\n", a, op, b, a / b);
        else
            printf("除数不能为0\n");
        break;
    default:
        printf("运算符错误\n");
        break;
    }
    return 0;
}

【题库1-第39题】
题目：用 switch 语句实现：接收用户输入的两个整数 x 和 y（x < y），输入 1 则计算 x 到 y 的和，输入 2 则计算 x 到 y 的平均值，输出结果。
答案：
#include <stdio.h>
int main(void)
{
    int x, y, choice, i;
    long long sum = 0;
    printf("请输入两个整数x和y（x<y）：");
    scanf("%d %d", &x, &y);
    printf("请输入选项（1：求和，2：求平均值）：");
    scanf("%d", &choice);
    switch (choice) {
    case 1:
        for (i = x; i <= y; i++)
            sum += i;
        printf("%d到%d的和为：%lld\n", x, y, sum);
        break;
    case 2:
        for (i = x; i <= y; i++)
            sum += i;
        printf("%d到%d的平均值为：%.2f\n", x, y, (double)sum / (y - x + 1));
        break;
    default:
        printf("选择错误\n");
        break;
    }
    return 0;
}
